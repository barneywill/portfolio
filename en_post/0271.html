MySQL 1 - Common commands

📅 January 15, 2019
📖 1 min read
👁️ 338 views

1 Create user

CREATE USER 'username'@'host' IDENTIFIED BY 'password';

for example

create user 'test_user'@'%' identified by 'test';

ps: If only local login is allowed, host=localhost, if login from any remote host is allowed, host=%

Change password

SET PASSWORD FOR 'username'@'host' = PASSWORD('newpassword');

2 Authorization

GRANT privileges ON dbname.tablename TO 'username'@'host'

for example

GRANT ALL ON test_db.* TO 'test_user'@'%';

revoke permission

REVOKE privilege ON dbname.tablename FROM 'username'@'host';

3 View the authorization of a user

show grants for test_user;

4 Data export

$ mysqldump - u dbuser - p dbname [tablename1] [tablename2] > dump.sql

ps: tablename is optional

If you want to export all databases, pass --all-databases to dataname.

More parameters:

--no-create-info Do not export table creation statements

--skip-add-locks Skip lock table statements

--where="..." Export only data that meets the conditions

5 Data import

mysql>source /path/dump.sql

or

$ mysql - uusername - p dbname < dump.sql

6 Positioning issues

mysql>show processlist;

mysql>show variables;

 

mysql>show index from $db.$table;

 

mysql>select @@profiling;

mysql>set profiling=1;

mysql>select * from $table;

mysql>show profiles;

mysql>show profile all for query $query_id;

mysql>set profiling=0;

7 Modify character set

mysql> alter database $database_name character set utf8;
mysql> alter table $table_name default character set utf8;
mysql> alter table $table_name change $column_name $column_name varchar(50) character utf8;
