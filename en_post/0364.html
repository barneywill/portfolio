  def main(args: Array[String]): Unit = {

    val appArgs = new SparkSubmitArguments(args) //parse command line parameter

    if (appArgs.verbose) {

      // scalastyle:off println

      // scalastyle:on println

    }

  private def runMain(

      childArgs: Seq[String],

      childClasspath: Seq[String],

    // scalastyle:off println

      printStream.println(s"Arguments:\n${childArgs.mkString("\n")}")

      printStream.println(s"System properties:\n${sysProps.mkString("\n")}")

Note: Here, the command line parameters, such as mainClass, are first parsed, and the running environment is prepared including System Property and classpath. Then a new classloader: ChildFirstURLClassLoader is used to load the user's mainClass, and then the main method of mainClass is called reflectively, so that the main method of the user's app.package.AppClass begins execution.

  private[spark] def loadFromSystemProperties(silent: Boolean): SparkConf = {

    // Load any spark.* system properties

      set(key, value, silent)
