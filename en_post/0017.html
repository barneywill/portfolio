Enabling LDAP Authentication in Hive

📅 May 20, 2022
📖 1 min read
👁️ 949 views

hive-site.xml

1 Access LDAP or ad domain

<property>
    <name>hive.server2.authentication</name>
    <value>LDAP</value>
</property>
<property>
    <name>hive.server2.authentication.ldap.url</name>
<value>ldap://test.com</value>
</property>
<property>
    <name>hive.server2.authentication.ldap.Domain</name>
    <value>test.com</value>
</property>

2 When executing SQL, use the login account instead of the Hive process to start the account

<property>
    <name>hive.server2.enable.doAs</name>
    <value>true</value>
</property>

3 Add account to admin

<property>
    <name>hive.users.in.admin.role</name>
    <value>username</value>
</property>

4 Change to SQL authorization

<property>
    <name>hive.security.authorization.manager</name>
    <value>org.apache.hadoop.hive.ql.security.authorization.plugin.sqlstd.SQLStdConfOnlyAuthorizerFactory</value>
</property>

After making the above changes, restart the Hive server.

core-site.xml

1 Allow Hive process startup account to switch to login account

<property>
   <name>hadoop.proxyuser.hive.hosts</name>
   <value>*</value>
</property>
<property>
   <name>hadoop.proxyuser.hive.groups</name>
   <value>*</value>
</property>

After making the above changes, restart the NameNode.
