View the logs (stdout and stderr) after starting the process

1 nohup+tail

# nohup $cmd > /path/to/file 2>&1 &
# tail - f /path/to/file

append
# nohup $cmd >> /path/to/file 2>&1 &

 

2 tee

# $cmd|& tee /path/to/file

append
# $cmd|& tee - a /path/to/file
