#!/bin/sh

1 Get script parameters

$# Number of parameters
$0 current script name
$1 1st parameter
$n nth parameter
$* all parameters
$@ all parameters
$? Status of the previous command
$$ current pid

Get current directory

$(cd `dirname $0`; pwd)

2 dates

$ date
Wed Mar 27 17:27:18 CST 2019
$ date +"%Y-%m-%d %H:%M:%S"
2019-03-27 17:28:27
$ date +"%Y-%m-%d %H:%M:%S" --date="-1 day"
2019-03-26 17:29:10
$ date +%s
1553678982

$ date - d "-1day2019-05-20" +%Y%m%d
20190519

$ date - d"$(date - d"1 month" +"%Y%m01") - 1 day" +"%Y%m%d"
20190531

$ date +"%Y%m$(cal|sed 'N;${s/.* //;P;d};D')"
20190531

3 String replacement

Replace all a in the variable var with b

${var//a/b}
echo "$var"|awk 'gsub("a","b") {print $0}'
echo "$"|sed 's/a/b/'

4 String interception substring

var=hello
echo ${#var}
echo $var|awk '{print substr($0, 2, 3)}'
echo ${var:2}
echo ${var:2:3}

echo ${var#he}
echo ${var%lo}

echo ${var1:`echo $((${#var2}+1))`}

5 string split

Array (the item itself cannot contain spaces)

str=${str//,/ }
for item in $str 
do
    echo $item
done

cut (cut: the delimiter must be a single character The delimiter can only be 1 character)

        i=1
        while((1==1))
        do
                item=`echo $str|cut - d "," -f$i`
                if [ "$item" != "" ]
                then
                        ((i++))
                        echo $item
                else
                        break
                fi
        done

awk - F (index position can only be hardcoded)

$ echo $str|awk -F ',' '{print $1}

awk split

$ echo $str|awk '{split($0,arr,","); for (i in arr) {print arr[i]}}'

self-increment id

echo "var is empty"

echo "var is not empty"
fi

if [[ "$var" = "" ]]
then
echo "empty"
fi

if [[ "$var" != "" ]]
then
echo "not empty"
fi

echo "a > 0"

echo "a < 0"
fi

13 File and folder judgment

if [ - d "/path/to/dir" ]
then
echo 'directory /path/to/dir exists'
else
echo 'directory /path/to/dir does not exists'
fi

if [ - f "/path/to/file" ]
then
echo 'file /path/to/file exists'
else
echo 'file /path/to/file does not exists'
fi

14 Traverse folders

dir=/path/to/dir/
for file in $dir/*
do
echo $file
done

15 Exit script

exit 0
